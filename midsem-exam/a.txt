<?php

/* A program to generate tokens and display the symbol table entry */

if (isset($_POST['submit']))
{
  $num = $_POST['num'];
  /* looping starts */
  for ($j=2;$j<=$num;$j++)
  {
    for($k=2;$k<$j;$k++)
    {
      if($j%$k==0)
      {
        break;
      }
      if( $k == $j )
        echo "Prime Number : $j <br />";
    }
  }
}
?>
